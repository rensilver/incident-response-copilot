import ast
import hashlib
import json
import os
import re
import stat
import subprocess
import tarfile
from pathlib import Path
from urllib.parse import unquote

def git(*args):
    return subprocess.check_output(["git", *args], text=True, timeout=30)

base = "0aa61cc"
start = "462f1f8dc04cd129e5c25cc4adba81c28a11fd0b"
tracked = git("ls-files").splitlines()
review = Path("docs/validation/2026-10-21-action1.2-review.md")
records = {str(review), "docs/validation/2026-10-21-action1.2-review.log"}
worktrees = git("worktree", "list", "--porcelain")
assert worktrees.count("worktree ") == 1 and "prunable" not in worktrees
assert f"worktree {Path.cwd()}\n" in worktrees
assert not git("worktree", "prune", "--dry-run", "--verbose").strip()
assert not git("fsck", "--connectivity-only", "--no-dangling").strip()
branch = "worktree-feature+evaluation-harness"
assert git("rev-parse", branch).strip() == "3ed9fcbaf407582237c1324e584866682497be0f"
git("merge-base", "--is-ancestor", branch, "HEAD")
assert git("rev-list", "--left-right", "--count", f"HEAD...{branch}").split()[1] == "0"
print("PASS: one worktree; no stale registrations; Git connectivity; retained ancestor branch")

archive_dir = Path("build/worktree-retirement/2026-10-21-evaluation-harness")
archive = archive_dir / "ignored-files.tar.gz"
manifest_path = archive_dir / "manifest.json"
manifest = json.loads(manifest_path.read_text())
assert manifest["worktree_head"] == git("rev-parse", branch).strip()
expected = "1d04caefde4a73abed758af459270d1a845f213a384566fc0a5e96647e4ce585"
assert hashlib.sha256(archive.read_bytes()).hexdigest() == manifest["archive_sha256"] == expected
assert stat.S_IMODE(archive_dir.stat().st_mode) == 0o700
assert all(stat.S_IMODE(p.stat().st_mode) == 0o600 for p in (archive, manifest_path))
with tarfile.open(archive, "r:gz") as bundle:
    members = bundle.getmembers()
    assert len(members) == len(manifest["preserved"]) == 13
    assert {m.name for m in members} == set(manifest["preserved"])
    for member in members:
        assert member.isfile()
        with bundle.extractfile(member) as stream:
            content = stream.read()
        entry = manifest["preserved"][member.name]
        assert len(content) == entry["size"]
        assert hashlib.sha256(content).hexdigest() == entry["sha256"]
print("PASS: recovery archive hash; 13 member sizes/hashes; private archive permissions")

removed = git("ls-tree", "-r", "--name-only", base, "CLAUDE.md", "docs/superpowers").splitlines()
assert len(removed) == 7

for name in removed:

    dirs[:] = [d for d in dirs if d not in excluded and not d.endswith(".egg-info")
               and Path(directory, d) != Path("docker/prometheus/data")]
    assert not any(tooling.search(name) for name in dirs + files), directory

links = 0
for name in sorted(set(tracked) | {str(review)}):
    source = Path(name)
    if source.suffix != ".md":
        continue
    prose = re.sub(r"^```[^\n]*\n.*?^```[^\n]*$", "", source.read_text(), flags=re.M | re.S)
    for target in re.findall(r"\]\(([^)]+)\)", prose):
        if re.match(r"[a-zA-Z][a-zA-Z0-9+.-]*:", target):
            continue

        resolved = source.parent / filename if filename else source
